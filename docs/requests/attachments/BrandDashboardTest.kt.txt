package com.aktcl.aron.backend.app

import com.fasterxml.jackson.databind.node.ObjectNode
import io.ktor.http.HttpStatusCode
import org.junit.jupiter.api.BeforeAll
import org.junit.jupiter.api.TestInstance
import kotlin.test.Test
import kotlin.test.assertEquals
import kotlin.test.assertTrue

/**
 * Attached to docs/requests/backend-authz-qa-1.md s1 (QA-FLOW-01). Add as backend/app/src/test/kotlin/.../BrandDashboardTest.kt with the fix (NationalDeps gets a BrandScopeResolver in Wiring.kt):
 * a BRAND user reads the product cards of its brands (docs/34 s1 level 4) and not the cards without a brand dimension.
 */
@TestInstance(TestInstance.Lifecycle.PER_CLASS)
class BrandDashboardTest : DataFlowTraceSupport() {
    override val traceName = "brand-dashboard"

    @BeforeAll
    fun prep() { prepareSrDay(); send(sr, oneSale("GUL2-001", 0, Line(pick("cigarette", 20), 3), "sr1001-261024-001"), uuid()); runWorker() }

    private val body get() = """{"date":"$day"}"""

    @Test
    fun aBrandUserReadsTheSalesCardsAndTheyAreASliceOfTheNationalFigures() {
        val brand = login("qabrand1001", "web", null)
        val r = postJson("/v1/dashboards/national/sales", brand, body)
        assertEquals(HttpStatusCode.OK, r.status)
        val national = cards(jsonOf(postJson("/v1/dashboards/national/sales", login("superadmin1001", "web", null), body)))
        val mine = cards(jsonOf(r))
        assertTrue(mine.all { (k, v) -> v <= (national[k] ?: 0L) }, "brand $mine vs national $national")
    }

    @Test
    fun cardsWithoutABrandDimensionAreRefused() {
        val brand = login("qabrand1001", "web", null)
        for (p in listOf("cpr", "final-submit", "login-submit", "geo-fencing")) assertEquals(HttpStatusCode.Forbidden, postJson("/v1/dashboards/national/$p", brand, body).status, p)
    }

    private fun cards(o: com.fasterxml.jackson.databind.JsonNode): Map<String, Long> = o.path("cards").associate { it.path("category_code").asText() to it.path("total_sales").asLong() }
}
